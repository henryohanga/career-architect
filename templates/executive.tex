% --- Executive Professional Style ---
% A sophisticated, authoritative resume template
% Best for: Senior roles, management, enterprise companies

\usepackage[empty]{fullpage}
\usepackage{titlesec}
\usepackage{xcolor}
\usepackage[unicode]{hyperref}
\usepackage{enumitem}
\usepackage{fontawesome5}
\usepackage{setspace}
\usepackage{parskip}

% Professional color scheme
\definecolor{navy}{HTML}{1E3A5F}
\definecolor{gold}{HTML}{B8860B}
\definecolor{charcoal}{HTML}{36454F}

% Executive margins - balanced white space
\addtolength{\oddsidemargin}{-0.35in}
\addtolength{\evensidemargin}{-0.35in}
\addtolength{\textwidth}{0.7in}
\addtolength{\topmargin}{-.35in}
\addtolength{\textheight}{0.7in}

% Distinguished section headers
\titleformat{\section}{
    \vspace{-3pt}\scshape\large\bfseries\color{navy}
}{}{0em}{}[\color{gold}\titlerule\vspace{-3pt}]

% Professional link styling
\hypersetup{colorlinks=true, urlcolor=navy, linkcolor=navy}

% Contact header is injected dynamically by scripts/build_resume.py
% via --include-before-body using identity.json values (only non-empty fields rendered).

% Refined bullet points
\setlist[itemize]{leftmargin=0.18in, labelsep=0.5em, nosep, itemsep=3pt}
\renewcommand{\labelitemi}{\small\textcolor{gold}{\faIcon{angle-right}}}

% Numbered list styling - professional
\setlist[enumerate]{leftmargin=0.22in, labelsep=0.5em, itemsep=3pt, topsep=4pt, parsep=2pt}
\setlist[enumerate,1]{label=\textcolor{navy}{\arabic*}.}
\setlist[enumerate,2]{label=\textcolor{charcoal}{\alph*}.}
\setlength{\parskip}{4pt}
\setstretch{1.15}

\pagenumbering{gobble}

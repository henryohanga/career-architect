% --- Minimal Clean Style ---
% A clean, distraction-free resume template
% Best for: Traditional industries, conservative companies

\usepackage[empty]{fullpage}
\usepackage{titlesec}
\usepackage[unicode]{hyperref}
\usepackage{enumitem}
\usepackage{fontawesome5}
\usepackage{setspace}
\usepackage{parskip}

% Minimal margins
\addtolength{\oddsidemargin}{-0.5in}
\addtolength{\evensidemargin}{-0.5in}
\addtolength{\textwidth}{1in}
\addtolength{\topmargin}{-.5in}
\addtolength{\textheight}{1in}

% Simple section headers - no color, clean lines
\titleformat{\section}{\vspace{-4pt}\large\bfseries\uppercase}{}{0em}{}[\titlerule\vspace{-4pt}]

% Subtle link styling
\hypersetup{colorlinks=true, urlcolor=black, linkcolor=black}

% Contact header is injected dynamically by scripts/build_resume.py
% via --include-before-body using identity.json values.

% Clean bullet points
\setlist[itemize]{leftmargin=0.2in, labelsep=0.4em, nosep, itemsep=2pt}
\renewcommand{\labelitemi}{\textbullet}

% Numbered list styling
\setlist[enumerate]{leftmargin=0.25in, labelsep=0.4em, itemsep=2pt, topsep=3pt, parsep=2pt}
\setlist[enumerate,1]{label=\arabic*.}
\setlist[enumerate,2]{label=\alph*., leftmargin=0.35in}
\setlength{\parskip}{3pt}

\pagenumbering{gobble}
